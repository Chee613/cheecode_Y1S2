array. With vertex order `A, B, C, D, E, F, G, H, I`:

## Question 3 - Linked-list representation

Use the **adjacency list**, since each vertex owns a linked list of its outgoing neighbours.

```java
import java.util.LinkedHashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;

public class Tutorial8Graph {
    public static void main(String[] args) {
        Map<String, List<String>> graph = new LinkedHashMap<>();
        for (String v : new String[]{"A","B","C","D","E","F","G","H","I"})
            graph.put(v, new LinkedList<>());

        addEdge(graph, "A", "C"); addEdge(graph, "A", "D");
        addEdge(graph, "B", "D"); addEdge(graph, "C", "E");
        addEdge(graph, "C", "F"); addEdge(graph, "D", "E");
        addEdge(graph, "E", "G"); addEdge(graph, "F", "H");
        addEdge(graph, "G", "H"); addEdge(graph, "H", "I");

        graph.forEach((v, neighbours) -> System.out.println(v + ": " + neighbours));
    }

    static void addEdge(Map<String, List<String>> graph, String from, String to) {
        graph.get(from).add(to);
    }
}
```
